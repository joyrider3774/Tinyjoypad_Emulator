cmake_minimum_required(VERSION 3.20)

project(Tinyjoypad_Emulator LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

# ---------------------------------------------------------------------------
# SDL3
#
# The SDL3 checkout at the repo root is a submodule, so vendoring is the
# default here (unlike a port that expects a system SDL).  Set
# -DUSE_VENDORED_SDL=OFF to link a system/installed SDL3 instead.
# ---------------------------------------------------------------------------
option(USE_VENDORED_SDL "Build the SDL3 checked out at ./SDL3 instead of finding an installed one" ON)

if(USE_VENDORED_SDL)
    if(NOT EXISTS "${PROJECT_SOURCE_DIR}/SDL3/CMakeLists.txt")
        message(FATAL_ERROR
            "USE_VENDORED_SDL is ON but ./SDL3 is empty. Run:\n"
            "  git submodule update --init --recursive")
    endif()
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON  CACHE BOOL "" FORCE)
    set(SDL_STATIC_PIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
    add_subdirectory("${PROJECT_SOURCE_DIR}/SDL3" "${CMAKE_BINARY_DIR}/SDL3" EXCLUDE_FROM_ALL)
    set(TJ_SDL_TARGET SDL3::SDL3-static)
else()
    find_package(SDL3 REQUIRED CONFIG COMPONENTS SDL3)
    set(TJ_SDL_TARGET SDL3::SDL3)
endif()

# ---------------------------------------------------------------------------
# simavr
#
# simavr ships a plain Makefile only, so its sources are compiled here
# directly.  Two pieces of the upstream build are replaced:
#
#   * sim_core_config.h / sim_core_decl.h, normally generated by probing
#     avr-gcc for every core, are checked in under simavr_config/ with only
#     the ATtiny85 enabled (see the comments in those files).
#   * libelf is not used.  sim_elf.c already degrades gracefully when
#     HAVE_LIBELF is undefined, and this emulator loads Intel HEX files via
#     read_ihex_file() anyway, so there is no external dependency at all.
# ---------------------------------------------------------------------------
set(SIMAVR_DIR "${PROJECT_SOURCE_DIR}/simavr/simavr")

if(NOT EXISTS "${SIMAVR_DIR}/sim/sim_avr.c")
    message(FATAL_ERROR
        "./simavr is empty. Run:\n"
        "  git submodule update --init --recursive")
endif()

file(GLOB SIMAVR_SIM_SRC CONFIGURE_DEPENDS
    "${SIMAVR_DIR}/sim/sim_*.c"
    "${SIMAVR_DIR}/sim/avr_*.c"
)
# run_avr.c is simavr's own command line front end - it has a main().
list(REMOVE_ITEM SIMAVR_SIM_SRC "${SIMAVR_DIR}/sim/run_avr.c")

add_library(simavr STATIC
    ${SIMAVR_SIM_SRC}
    "${SIMAVR_DIR}/cores/sim_tiny85.c"
    "${SIMAVR_DIR}/cores/sim_tinyx5.c"
)

target_include_directories(simavr PUBLIC
    "${PROJECT_SOURCE_DIR}/simavr_config"
    "${SIMAVR_DIR}"
    "${SIMAVR_DIR}/sim"
    "${SIMAVR_DIR}/cores"
)

# simavr's own warning hygiene is not this project's business.
if(MSVC)
    target_compile_options(simavr PRIVATE /w)
else()
    target_compile_options(simavr PRIVATE -w)
endif()

if(WIN32)
    # sim_gdb.c speaks winsock when built with MinGW.
    target_link_libraries(simavr PUBLIC ws2_32)
endif()

# ---------------------------------------------------------------------------
# The emulator itself
# ---------------------------------------------------------------------------
file(GLOB TJ_SRC CONFIGURE_DEPENDS
    "${PROJECT_SOURCE_DIR}/src/*.c"
    "${PROJECT_SOURCE_DIR}/src/*.h"
)

add_executable(${PROJECT_NAME} ${TJ_SRC})

target_include_directories(${PROJECT_NAME} PRIVATE "${PROJECT_SOURCE_DIR}/src")
target_link_libraries(${PROJECT_NAME} PRIVATE ${TJ_SDL_TARGET} simavr)

if(NOT MSVC)
    target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()

if(WIN32)
    # No console window on Windows; SDL_main routes WinMain to main().
    set_target_properties(${PROJECT_NAME} PROPERTIES WIN32_EXECUTABLE TRUE)
    if(MINGW)
        target_link_options(${PROJECT_NAME} PRIVATE
            -static-libgcc -static
        )
    endif()
endif()
